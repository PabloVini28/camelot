40def adaptive_threshold( 

41 imagename, process_background=False, blocksize=15, c=-2, rotation="" 

42): 

43 """Thresholds an image using OpenCV's adaptiveThreshold. 

44 

45 Parameters 

46 ---------- 

47 imagename : str or numpy.ndarray 

48 Path to an image file, or an already-decoded BGR image array (the 

49 latter lets callers skip the PNG round-trip — see #40). 

50 process_background : bool, optional (default: False) 

51 Whether or not to process lines that are in background. 

52 blocksize : int, optional (default: 15) 

53 Size of a pixel neighborhood that is used to calculate a 

54 threshold value for the pixel: 3, 5, 7, and so on. 

55 

56 For more information, refer `OpenCV's adaptiveThreshold 

57 <https://docs.opencv.org/2.4/modules/imgproc/doc/miscellaneous_transformations.html#adaptivethreshold>`_. 

58 c : int, optional (default: -2) 

59 Constant subtracted from the mean or weighted mean. 

60 Normally, it is positive but may be zero or negative as well. 

61 

62 For more information, refer `OpenCV's adaptiveThreshold 

63 <https://docs.opencv.org/2.4/modules/imgproc/doc/miscellaneous_transformations.html#adaptivethreshold>`_. 

64 rotation: str, optional (default: "") 

65 Either "" (no rotation), "clockwise", or "anticlockwise". The 

66 **inverse** of this rotation will be applied to the image. 

67 

68 Returns 

69 ------- 

70 img : object 

71 numpy.ndarray representing the original image. 

72 threshold : object 

73 numpy.ndarray representing the thresholded image. 

74 """ 

75 img = imagename if isinstance(imagename, np.ndarray) else cv2.imread(imagename) 

76 img = undo_rotation(img, rotation) 

77 gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) 

78 if not process_background: 

79 gray = np.invert(gray) 

80 threshold = cv2.adaptiveThreshold( 

81 gray, 255, cv2.ADAPTIVE_THRESH_GAUSSIAN_C, cv2.THRESH_BINARY, blocksize, c 

82 ) 

83 return img, threshold

202def process_image(threshold, el, iterations, erode_iterations=0): 

203 """ 

204 Apply morphological operations to the threshold image. 

205 

206 Parameters 

207 ---------- 

208 threshold : object 

209 numpy.ndarray representing the thresholded image. 

210 el : object 

211 Structuring element for morphological operations. 

212 iterations : int 

213 Number of dilation passes applied to close small gaps in the 

214 line mask. 

215 erode_iterations : int, optional (default: 0) 

216 Number of erosion passes applied *after* the dilation. When 

217 equal to ``iterations`` this is a morphological closing — 

218 gaps in the lines are bridged without thickening the mask

220 

221 Returns 

222 ------- 

223 numpy.ndarray 

224 The processed threshold image. 

225 """ 

226 threshold = cv2.erode(threshold, el)

229 if erode_iterations: 229 ↛ 230line 229 didn't jump to line 230 because the condition on line 229 was never true

230 dmask = cv2.erode(dmask, el, iterations=erode_iterations) 

231 

232 return dmask

235def extract_lines_from_contours(contours, direction): 

236 """ 

237 Extract lines from contours based on the specified direction. 

238 

239 Parameters 

240 ---------- 

241 contours : list 

242 List of contours found in the image. 

243 direction : string 

244 Specifies whether to extract vertical or horizontal lines. 

245 

246 Returns 

247 ------- 

248 list 

249 List of tuples representing the coordinates of the lines.

254 x1, x2 = x, x + w

264def find_contours(vertical, horizontal): 

265 """Find table boundaries using OpenCV's findContours. 

266 

267 Parameters 

268 ---------- 

269 vertical : object 

270 numpy.ndarray representing pixels where vertical lines lie. 

271 horizontal : object 

272 numpy.ndarray representing pixels where horizontal lines lie. 

273 

274 Returns 

275 ------- 

276 cont : list 

277 List of tuples representing table boundaries. Each tuple is of 

278 the form (x, y, w, h) where (x, y) -> left-top, w -> width and 

279 h -> height in image coordinate space. 

280 

281 """ 

282 mask = vertical + horizontal 

283 

284 contours, __ = cv2.findContours( 

285 mask.astype(np.uint8), cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE 

286 ) 

287 # Sort largest-first so callers iterating in detected order pick up the 

288 # primary table first; filter out anything smaller than the configured 

289 # fraction of the page area. Replaces the previous arbitrary 10-contour 

290 # cap that silently dropped tables past index 9 (#319). 

291 page_area = float(mask.shape[0] * mask.shape[1]) or 1.0 

292 min_area = page_area * _MIN_TABLE_AREA_FRACTION 

293 contours = sorted(contours, key=cv2.contourArea, reverse=True) 

294 contours = [c for c in contours if cv2.contourArea(c) >= min_area] 

295 

296 cont = [] 

297 for c in contours: 

298 c_poly = cv2.approxPolyDP(c, 3, True) 

299 x, y, w, h = cv2.boundingRect(c_poly) 

300 cont.append((x, y, w, h)) 

301 return cont

304def find_joints(contours, vertical, horizontal): 

305 """Find joints/intersections present inside each table boundary. 

306 

307 Parameters 

308 ---------- 

309 contours : list 

310 List of tuples representing table boundaries. Each tuple is of 

311 the form (x, y, w, h) where (x, y) -> left-top, w -> width and 

312 h -> height in image coordinate space. 

313 vertical : object 

314 numpy.ndarray representing pixels where vertical lines lie. 

315 horizontal : object 

316 numpy.ndarray representing pixels where horizontal lines lie. 

317 

318 Returns 

319 ------- 

320 tables : dict 

321 Dict with table boundaries as keys and list of intersections 

322 in that boundary as their value. 

323 Keys are of the form (x1, y1, x2, y2) where (x1, y1) -> lb 

324 and (x2, y2) -> rt in image coordinate space. 

325 

326 """ 

327 joints = np.multiply(vertical, horizontal) 

328 tables = {} 

329 for c in contours: 

330 x, y, w, h = c 

331 roi = joints[y : y + h, x : x + w] 

332 jc, __ = cv2.findContours( 

333 roi.astype(np.uint8), cv2.RETR_CCOMP, cv2.CHAIN_APPROX_SIMPLE 

334 ) 

335 if len(jc) <= 4: # remove contours with less than 4 joints 

336 continue 

337 joint_coords = [] 

338 for j in jc: 

339 jx, jy, jw, jh = cv2.boundingRect(j) 

340 c1, c2 = x + (2 * jx + jw) // 2, y + (2 * jy + jh) // 2 

341 joint_coords.append((c1, c2)) 

342 tables[(x, y + h, x + w, y)] = joint_coords 

343 

344 return tables

347# --- Stage 1 of #763: vector-line engine for Lattice ------------------------- 

348# 

349# These helpers read ruled lines directly from playa's layout tree (LTLine / 

350# LTRect) instead of rasterising the page and using OpenCV findContours. They 

351# are not wired into the Lattice parser yet — that's Stage 2 of #763, gated 

352# behind a new ``engine='vector'`` kwarg. Stage 1 lands the helpers + tests 

353# so the function shape can be reviewed in isolation. 

354 

355#: Angle tolerance (in PDF units) within which a line is considered 

356#: orthogonal. PDFs draw nominally-axis-aligned ruled lines with the exact 

357#: equal coordinates we want, but a tiny float epsilon — be generous, we 

358#: don't want to drop a 0.0001-unit-off line just because of rounding. 

359_LINE_ORTHOGONAL_TOL = 0.5 

360 

361#: Tolerance (PDF units) for merging near-collinear ruled-line segments. 

362#: PDFs routinely draw one logical rule as many short, abutting sub-segments; 

363#: segments whose constant axis agrees within this tolerance and whose extents 

364#: overlap or sit within it of each other are fused into a single line. 

365_LINE_COALESCE_TOL = 2.0 

366 

367#: A "filled" rectangle whose narrower side is below this width is treated 

368#: as a stroked line (some PDF generators draw ruled lines as 0.5pt-wide 

369#: filled rectangles rather than as stroked LTLines). 

370_LINE_AS_THIN_RECT_TOL = 1.5

381def _ruled_lines_from_layout(layout): 

382 """Collect ruled line segments from a layout tree (PDF coord space). 

383 

384 Returns a list of ``(x0, y0, x1, y1)`` tuples. Both stroked LTLines 

385 and the four edges of stroked LTRects are emitted. Filled-but-not- 

386 stroked LTRects with one narrow dimension are also treated as lines 

387 (covers PDFs that draw rules as thin filled rectangles). 

388 """ 

389 out: list[tuple[float, float, float, float]] = [] 

390 for obj in _walk_layout_objects(layout): 

391 if isinstance(obj, LTLine) and getattr(obj, "stroke", True): 

392 out.append((obj.x0, obj.y0, obj.x1, obj.y1)) 

393 elif isinstance(obj, LTRect): 

394 stroked = getattr(obj, "stroke", False) 

395 filled = getattr(obj, "fill", False) 

396 if stroked: 

397 # Four edges of the rectangle as separate line segments. 

398 out.append((obj.x0, obj.y0, obj.x1, obj.y0)) # bottom 

399 out.append((obj.x0, obj.y1, obj.x1, obj.y1)) # top 

400 out.append((obj.x0, obj.y0, obj.x0, obj.y1)) # left 

401 out.append((obj.x1, obj.y0, obj.x1, obj.y1)) # right 

402 elif filled: 

403 # Thin filled rect → treat the long axis as the line. 

404 width = obj.x1 - obj.x0 

405 height = obj.y1 - obj.y0 

406 if ( 

407 min(width, height) <= _LINE_AS_THIN_RECT_TOL 

408 and max(width, height) > _LINE_AS_THIN_RECT_TOL 

409 ): 

410 if width >= height: 

411 mid_y = (obj.y0 + obj.y1) / 2 

412 out.append((obj.x0, mid_y, obj.x1, mid_y)) 

413 else: 

414 mid_x = (obj.x0 + obj.x1) / 2 

415 out.append((mid_x, obj.y0, mid_x, obj.y1)) 

416 return out

419def find_lines_from_layout(layout, direction="horizontal"): 

420 """Return ruled lines from a playa layout tree (Stage 1 of #763). 

421 

422 Drop-in companion to :func:`find_lines` for ``flavor='lattice'``'s 

423 vector engine. Walks ``layout`` (recursively), classifies each 

424 LTLine / stroked LTRect / thin filled LTRect into horizontal or 

425 vertical based on its dominant axis, and returns the subset matching 

426 ``direction``. 

427 

428 Parameters 

429 ---------- 

430 layout : object 

431 An ``LTPage`` / ``LTContainer`` from ``playa`` — typically what 

432 the lattice parser already stores on ``self.layout``. 

433 direction : str, optional (default: 'horizontal') 

434 ``'horizontal'`` returns lines whose y-delta is below the 

435 orthogonality tolerance; ``'vertical'`` returns those whose 

436 x-delta is below it. 

437 

438 Returns 

439 ------- 

440 lines : list[tuple[float, float, float, float]] 

441 Each tuple is ``(x0, y0, x1, y1)`` in PDF coordinate space — 

442 same shape as :func:`find_lines`'s second return value, but in 

443 PDF coords (not image coords). The Lattice integration in 

444 Stage 2 will apply the existing ``scale_pdf`` to convert. 

445 

446 Notes 

447 ----- 

448 The output drops the morphological mask :func:`find_lines` produces 

449 as its first return; the vector engine doesn't need it because it 

450 bypasses :func:`find_contours` (which is the only consumer of the 

451 mask) and computes table bboxes from line intersections directly. 

452 """ 

453 if direction not in ("horizontal", "vertical"): 

454 raise ValueError("direction must be 'horizontal' or 'vertical'") 

455 raw = _ruled_lines_from_layout(layout) 

456 result: list[tuple[float, float, float, float]] = [] 

457 for x0, y0, x1, y1 in raw: 

458 dx = abs(x1 - x0) 

459 dy = abs(y1 - y0) 

460 if direction == "horizontal" and dy <= _LINE_ORTHOGONAL_TOL and dx > 0: 

461 result.append((min(x0, x1), y0, max(x0, x1), y0)) 

462 elif direction == "vertical" and dx <= _LINE_ORTHOGONAL_TOL and dy > 0: 

463 result.append((x0, min(y0, y1), x0, max(y0, y1))) 

464 return coalesce_collinear_lines(result, direction)

467def coalesce_collinear_lines(lines, direction, tol=_LINE_COALESCE_TOL): 

468 """Merge near-collinear ruled-line segments into single lines. 

469 

470 PDFs frequently draw one logical rule as a chain of short sub-segments; 

471 left unmerged they make :func:`find_joints_from_lines` under-count 

472 joints and miss grids. Segments are grouped by their constant axis 

473 (within ``tol``) and, within a group, those whose extents overlap or 

474 sit within ``tol`` of each other are fused. 

475 

476 Parameters 

477 ---------- 

478 lines : list[tuple[float, float, float, float]] 

479 ``(x0, y0, x1, y1)`` segments, as produced by 

480 :func:`find_lines_from_layout`. 

481 direction : str 

482 ``'horizontal'`` (segments share a constant ``y``, merge along 

483 ``x``) or ``'vertical'`` (constant ``x``, merge along ``y``). 

484 tol : float, optional 

485 Merge tolerance in PDF units (default :data:`_LINE_COALESCE_TOL`). 

486 

487 Returns 

488 ------- 

489 list[tuple[float, float, float, float]] 

490 The merged segments, same tuple shape as the input. 

491 """ 

492 if direction not in ("horizontal", "vertical"): 

493 raise ValueError("direction must be 'horizontal' or 'vertical'") 

494 if not lines: 

495 return [] 

496 

497 norm = _coalesce_normalise(lines, direction) 

498 

499 # Chain into clusters of (nearly) equal constant axis. 

500 clusters: list[dict] = [] 

501 for const, lo, hi in norm: 

502 if clusters and const - clusters[-1]["last"] <= tol: 

503 clusters[-1]["consts"].append(const) 

504 clusters[-1]["intervals"].append((lo, hi)) 

505 clusters[-1]["last"] = const 

506 else: 

507 clusters.append({"consts": [const], "intervals": [(lo, hi)], "last": const}) 

508 

509 out = [] 

510 for cluster in clusters: 

511 const = sum(cluster["consts"]) / len(cluster["consts"]) 

512 for lo, hi in _merge_spans(sorted(cluster["intervals"]), tol): 

513 out.append( 

514 (lo, const, hi, const) 

515 if direction == "horizontal" 

516 else (const, lo, const, hi) 

517 ) 

518 return out

521def _coalesce_normalise(lines, direction): 

522 """Reduce each segment to ``(const_axis, span_lo, span_hi)``, sorted.""" 

523 norm = [] 

524 for x0, y0, x1, y1 in lines: 

525 if direction == "horizontal": 

526 const = (y0 + y1) / 2.0 

527 lo, hi = (x0, x1) if x0 <= x1 else (x1, x0) 

528 else: 

529 const = (x0 + x1) / 2.0 

530 lo, hi = (y0, y1) if y0 <= y1 else (y1, y0) 

531 norm.append((const, lo, hi)) 

532 norm.sort() 

533 return norm

536def _merge_spans(intervals, tol): 

537 """Merge overlapping / within-``tol`` ``(lo, hi)`` intervals.""" 

538 merged = [list(intervals[0])] 

539 for lo, hi in intervals[1:]: 

540 if lo <= merged[-1][1] + tol: 

541 merged[-1][1] = max(merged[-1][1], hi) 

542 else: 

543 merged.append([lo, hi]) 

544 return merged

548#: vertical ruled line "cross" — accounts for lines that stop a hair short

550_JOINT_TOL = 2.0

555_MIN_JOINTS = 4

558def _line_crossing(h_line, v_line, tol=_JOINT_TOL): 

559 """Return the (x, y) crossing of a horizontal and vertical line, or None. 

560 

561 ``h_line`` is ``(x0, y, x1, y)`` (constant y), ``v_line`` is 

562 ``(x, y0, x, y1)`` (constant x), both in PDF coords as produced by 

563 :func:`find_lines_from_layout`. They cross when the vertical line's x 

564 falls within the horizontal line's x-span and the horizontal line's y 

565 falls within the vertical line's y-span, each within ``tol``. 

566 """ 

567 hx0, hy, hx1, _ = h_line 

568 vx, vy0, _, vy1 = v_line 

569 x_lo, x_hi = (hx0, hx1) if hx0 <= hx1 else (hx1, hx0) 

570 y_lo, y_hi = (vy0, vy1) if vy0 <= vy1 else (vy1, vy0) 

571 if (x_lo - tol) <= vx <= (x_hi + tol) and (y_lo - tol) <= hy <= (y_hi + tol): 

572 return (vx, hy) 

573 return None

576class _UnionFind: 

577 """Minimal union-find with path-halving, for clustering crossing lines.""" 

578 

579 def __init__(self, n): 

580 self.parent = list(range(n)) 

581 

582 def find(self, i): 

583 while self.parent[i] != i: 

584 self.parent[i] = self.parent[self.parent[i]] 

585 i = self.parent[i] 

586 return i 

587 

588 def union(self, a, b): 

589 ra, rb = self.find(a), self.find(b) 

590 if ra != rb: 

591 self.parent[ra] = rb

594def _line_crossings(horizontal_lines, vertical_lines, tol): 

595 """Union-find over crossing h/v lines. 

596 

597 Returns ``(uf, crossings)`` where ``uf`` is a :class:`_UnionFind` 

598 spanning ``[0, n_h)`` horizontal + ``[n_h, n_h+n_v)`` vertical line 

599 indices, joined wherever an h-line crosses a v-line, and 

600 ``crossings`` maps ``(h_idx, v_idx)`` to the ``(x, y)`` crossing. 

601 """ 

602 n_h = len(horizontal_lines) 

603 uf = _UnionFind(n_h + len(vertical_lines))

623def _cluster_lines(horizontal_lines, vertical_lines, tol=_JOINT_TOL): 

624 """Group mutually-intersecting lines into table clusters (union-find). 

625 

626 Returns a list of ``(bbox, joints)`` tuples — one per connected 

627 component of crossing lines — where ``bbox`` is ``(x0, y0, x1, y1)`` 

628 (left, bottom, right, top in PDF coords) and ``joints`` is the list of 

629 ``(x, y)`` crossing points within that component. Clusters with 

630 ``<= _MIN_JOINTS`` crossings are dropped (noise / stray L-corners), 

631 matching the raster pipeline's joint-count filter. 

632 

633 This is the vector-native replacement for the raster 

634 ``find_contours`` + ``find_joints`` pair: instead of one bbox over the 

635 union of every line (the Stage-1 prototype's flaw, which merged 

636 separate tables), it isolates each grid as its own component, so 

637 multi-table pages yield multiple bboxes. 

638 """ 

639 n_h = len(horizontal_lines) 

640 uf, crossings = _line_crossings(horizontal_lines, vertical_lines, tol) 

641 

642 comp_h = {} 

643 comp_v = {} 

644 comp_joints = {} 

645 for hi in range(n_h): 

646 comp_h.setdefault(uf.find(hi), []).append(horizontal_lines[hi]) 

647 for vi in range(len(vertical_lines)): 

648 comp_v.setdefault(uf.find(n_h + vi), []).append(vertical_lines[vi]) 

649 for (hi, _vi), pt in crossings.items(): 

650 comp_joints.setdefault(uf.find(hi), []).append(pt) 

651 

652 clusters = [] 

653 for root, joints in comp_joints.items(): 

654 if len(joints) <= _MIN_JOINTS: 

655 continue 

656 cluster = _cluster_bbox(comp_h.get(root, []), comp_v.get(root, []), joints) 

657 if cluster is not None: 657 ↛ 653line 657 didn't jump to line 653 because the condition on line 657 was always true

658 clusters.append(cluster) 

659 return clusters

662def find_contours_from_lines(horizontal_lines, vertical_lines, tol=_JOINT_TOL): 

663 """Vector-native table-bbox detection (Stage 2 of #763). 

664 

665 Companion to the raster :func:`find_contours`. Given ruled lines in 

666 PDF coords (from :func:`find_lines_from_layout`), returns a list of 

667 ``(x0, y0, x1, y1)`` table bounding boxes — one per connected cluster 

668 of mutually-intersecting lines — so a page with two separate tables 

669 yields two bboxes (the Stage-1 prototype merged them). 

670 

671 Returns 

672 ------- 

673 list[tuple[float, float, float, float]] 

674 ``(x0, y0, x1, y1)`` = (left, bottom, right, top) in PDF coords, 

675 sorted top-to-bottom (descending y0) to match reading order. 

676 """ 

677 clusters = _cluster_lines(horizontal_lines, vertical_lines, tol) 

678 bboxes = [bbox for bbox, _ in clusters] 

679 bboxes.sort(key=lambda b: -b[1]) # top table first 

680 return bboxes
