101def _resolve_filepath(filepath: FilepathOrBuffer) -> tuple[str | Path, bool]: 

102 """Normalise read_pdf's polymorphic filepath argument to a real path. 

103 

104 Returns ``(path, is_temp)``. ``is_temp`` is True when the returned 

105 path points to a tempfile we created (and therefore own — the 

106 caller's :meth:`PDFHandler.close` should reap it). 

107 

108 Three branches: 

109 

110 * ``bytes`` / ``bytearray`` / ``memoryview``: spill to a tempfile. 

111 * binary file-like (``.read()`` returning bytes): read once, 

112 preserve the caller's cursor, spill to a tempfile. 

113 * everything else (path / URL): URL → download to tempfile; path 

114 → pass through untouched. 

115 """ 

116 if isinstance(filepath, (bytes, bytearray, memoryview)): 

117 return _spill_bytes_to_tempfile(bytes(filepath)), True 

118 if hasattr(filepath, "read") and callable(filepath.read): 

119 data = _consume_filelike_to_bytes(filepath) # type: ignore[arg-type] 

120 return _spill_bytes_to_tempfile(data), True 

121 if is_url(filepath): 

122 return download_url(str(filepath)), True 

123 # mypy: filepath is the str | Path | os.PathLike subset of 

124 # FilepathOrBuffer (we've ruled out bytes / file-like / URL). 

125 return filepath, False # type: ignore[return-value]
